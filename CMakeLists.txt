cmake_minimum_required(VERSION 3.21)
project(QChat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent DBus LinguistTools)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GPGME REQUIRED IMPORTED_TARGET gpgme)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
add_library(qchat_crypto src/crypto.cpp)
target_include_directories(qchat_crypto PUBLIC src)
target_link_libraries(qchat_crypto PUBLIC Qt6::Core PkgConfig::GPGME)
target_compile_options(qchat_crypto PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(qchat src/main.cpp src/window.cpp src/window.h src/systemwatch.cpp src/systemwatch.h)
target_link_libraries(qchat PRIVATE qchat_crypto Qt6::Widgets Qt6::Concurrent Qt6::DBus PkgConfig::UDEV)
target_compile_options(qchat PRIVATE -Wall -Wextra -Wpedantic)
qt_add_translations(qchat TS_FILES i18n/qchat_de.ts i18n/qchat_fa.ts RESOURCE_PREFIX /i18n)
install(TARGETS qchat RUNTIME DESTINATION bin)
install(FILES packaging/org.qchat.QChat.desktop DESTINATION share/applications)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(crypto_test tests/crypto_test.cpp)
 target_link_libraries(crypto_test PRIVATE qchat_crypto Qt6::Test)
 add_test(NAME crypto_roundtrip COMMAND crypto_test)
 set_tests_properties(crypto_roundtrip PROPERTIES TIMEOUT 120)
 add_executable(window_test tests/window_test.cpp src/window.cpp src/window.h src/systemwatch.cpp src/systemwatch.h)
 target_link_libraries(window_test PRIVATE qchat_crypto Qt6::Test Qt6::Widgets Qt6::Concurrent Qt6::DBus PkgConfig::UDEV)
 add_dependencies(window_test qchat)
 add_test(NAME ui_workflows COMMAND window_test)
 set_tests_properties(ui_workflows PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QCHAT_TEST_ARTIFACTS=${CMAKE_BINARY_DIR}/artifacts" TIMEOUT 120)
 add_executable(theme_test tests/theme_test.cpp src/window.cpp src/window.h src/systemwatch.cpp src/systemwatch.h)
 target_link_libraries(theme_test PRIVATE qchat_crypto Qt6::Test Qt6::Widgets Qt6::Concurrent Qt6::DBus PkgConfig::UDEV)
 add_test(NAME theme_contrast COMMAND theme_test)
 set_tests_properties(theme_contrast PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QCHAT_TEST_ARTIFACTS=${CMAKE_BINARY_DIR}/artifacts" TIMEOUT 20)
 add_test(NAME ui_smoke COMMAND qchat --smoke-test)
 set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
endif()
